<objective>
Fix the cross-zone exit load-order bug (BLD-06). Currently _load_all_zones() does a single pass and silently fails when zone A references zone B:room_id but zone B hasn't loaded yet. This plan implements a two-pass strategy: first pass loads all zones, second pass retries all unresolved cross-zone exits from the first pass.

Purpose: Without this fix, any cross-zone exit defined before its target zone loads is silently dropped. All zone content authored through the builder relies on deterministic cross-zone connectivity.
Output: Modified area_builder.py (expose_unresolved_exits) + modified at_server_startstop.py (two-pass _load_all_zones)
</objective>

```python
def _resolve_cross_zone_exits(self):
    """Resolve deferred cross-zone exits by tag lookup."""
    still_deferred = []
    for exit_data in self._deferred_exits:
        target_str = exit_data["to"]   # peek, don't pop yet
        target_zone_id, target_room_id = target_str.split(":", 1)

        candidates = evennia.search_tag(target_room_id, category="room_id")
        target = None
        for room in candidates:
            if (room.db.zone_id or "") == target_zone_id:
                target = room
                break

        if not target:
            self._build_warnings.append(
                f"Cross-zone exit unresolved: {target_str} "
                f"(target zone may not be loaded yet)"
            )
            # Save for retry — keep full dict intact
            self._unresolved_exits.append(dict(exit_data))
            continue

        # Resolved — build the exit
        resolved = dict(exit_data)
        resolved.pop("to")
        from_room = resolved.pop("from_room")
        direction = resolved.pop("direction")
        self._create_exit_object(from_room, target, direction, **resolved)

    self._deferred_exits = still_deferred
```

```python
def _load_all_zones():
    """
    Load all area files from world/areas/ on server start.
    Idempotent — safe to call on restart.

    Two-pass loading (BLD-06):
      Pass 1 — all zones build normally; cross-zone exits may fail if
               the target zone hasn't loaded yet.
      Pass 2 — retry all unresolved cross-zone exits now that all zones
               are in DB.
    """
    import os
    import importlib
    from django.conf import settings
    from world import zone_registry, named_mob_registry
    import evennia as _evennia

    # Clear registries before rebuild
    zone_registry.clear()
    named_mob_registry.clear()

    areas_dir = os.path.join(settings.GAME_DIR, "world", "areas")
    if not os.path.exists(areas_dir):
        return

    # --- Pass 1: load all .py zone files ---
    builders = []  # retain builders for second-pass retry
    for filename in sorted(os.listdir(areas_dir)):
        if filename.endswith(".py") and not filename.startswith("_"):
            module_name = f"world.areas.{filename[:-3]}"
            try:
                if module_name in importlib.sys.modules:
                    importlib.reload(importlib.sys.modules[module_name])
                else:
                    importlib.import_module(module_name)
                module = importlib.sys.modules[module_name]
                if hasattr(module, "build"):
                    report = module.build()
                    # Retain the builder instance for second-pass retry
                    if hasattr(module, "_area_builder"):
                        builders.append(module._area_builder)
                    warnings = report.get("warnings", [])
                    for w in warnings:
                        # Skip "may not be loaded yet" — handled in pass 2
                        if "target zone may not be loaded yet" not in w:
                            print(f"Zone warning [{filename}]: {w}")
                    unresolved = report.get("unresolved_exits", [])
                    if unresolved:
                        print(f"Zone [{filename}]: {len(unresolved)} cross-zone exit(s) deferred to second pass")
            except Exception as e:
                import traceback
                print(f"Error loading zone {filename}: {e}")
                traceback.print_exc()

    # --- Pass 2: retry all unresolved cross-zone exits ---
    # Collect unresolved exits from all builders
    all_unresolved = []
    for builder in builders:
        if hasattr(builder, "expose_unresolved_exits"):
            all_unresolved.extend(builder.expose_unresolved_exits())

    if not all_unresolved:
        return

    resolved_count = 0
    still_unresolved = 0
    for exit_data in all_unresolved:
        target_str = exit_data["to"]
        target_zone_id, target_room_id = target_str.split(":", 1)

        candidates = _evennia.search_tag(target_room_id, category="room_id")
        target = None
        for room in candidates:
            if (room.db.zone_id or "") == target_zone_id:
                target = room
                break

        if not target:
            print(f"Cross-zone exit STILL unresolved after second pass: {target_str} (zone not loaded)")
            still_unresolved += 1
            continue

        from_room = exit_data["from_room"]
        direction = exit_data["direction"]
        # Collect remaining kwargs (exclude routing keys)
        kwargs = {k: v for k, v in exit_data.items()
                  if k not in ("from_room", "to", "direction")}
        try:
            from world.area_builder import AreaBuilder
            # Use a minimal builder just to call _create_exit_object
            _retry_builder = AreaBuilder.__new__(AreaBuilder)
            _retry_builder._zone_id = from_room.db.zone_id or "unknown"
            _retry_builder._exits_created = 0
            _retry_builder._create_exit_object(from_room, target, direction, **kwargs)
            resolved_count += 1
        except Exception as e:
            print(f"Error resolving cross-zone exit {target_str}: {e}")
            still_unresolved += 1

    print(f"Second pass: resolved {resolved_count} cross-zone exit(s), {still_unresolved} still unresolved")
```
