```python
def execute_action(action_dict, context, _depth=0):
    """
    Execute one action from the shared vocabulary.
    context keys: character, room, mob, item (all optional but relevant handler requires them).
    _depth: recursion depth for trigger chaining — blocked at >= 3 (D-19).
    Returns (bool, str).
    """
    if _depth >= 3:
        return False, "Trigger chain depth limit reached."
    action_type = action_dict.get("action_type")
    handler = ACTION_HANDLERS.get(action_type)
    if not handler:
        return False, f"Unknown action type: {action_type}"
    return handler(action_dict, context, _depth)
```

```python
def _handle_echo(action_dict, context, _depth):

    character.msg(msg)
    return True, ""
```

4. Public entry:
```python
def fire_triggers(source_obj, event_name, character, context=None, _depth=0):
    """
    Fire all triggers on source_obj matching event_name for character.
    source_obj: room, mob, or item with db.triggers list.
    event_name: "on_enter", "on_exit", "on_first_visit", "on_mob_death", "on_examine"
    character: the Character triggering the event.
    _depth: recursion counter for D-19 chaining limit.
    """
    if not hasattr(character, 'account') or not character.account:
        return  # Guard: only fire for actual player characters (Pitfall 7)
    triggers = list(source_obj.db.triggers or [])
    ctx = context or {}
    ctx.update({"character": character, "room": character.location, "mob": source_obj if hasattr(source_obj, 'db') and source_obj.db.mob_type else None})
    for trigger in triggers:
        if trigger.get("event") != event_name:
            continue
        trigger_id = trigger.get("trigger_id", "")
        if _check_once_per(character, trigger_id, trigger):
            continue
        if _check_cooldown(character, trigger_id, trigger):
            continue
        for action in trigger.get("actions", []):
            from world.action_vocabulary import execute_action
            execute_action(action, ctx, _depth=_depth)
        _record_fired(character, trigger_id, trigger)
```

5. on_first_visit logic: check `character.db.fired_triggers` for a sentinel `"visited:{room.id}"`. If not present, fire "on_first_visit" triggers and record the sentinel.

Note: `character.db.fired_triggers` initializes as an empty set `set()` if None; `character.db.trigger_cooldowns` initializes as an empty dict `{}` if None. Always initialize before use.
