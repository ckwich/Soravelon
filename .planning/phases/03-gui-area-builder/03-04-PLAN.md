class ZoneSerializerTestBase(EvenniaTest):
    """Shared setup for zone serializer tests."""

    def setUp(self):
        super().setUp()
        zone_registry.clear()
        named_mob_registry.clear()
        clear_unresolved_exits()

    def _minimal_zone_data(self, zone_id="ser_test_zone", **zone_overrides):
        """Return a minimal valid zone_data dict."""
        zone_meta = {
            "zone_id": zone_id,
            "name": "Serializer Test Zone",
            "zone_type": "plains",
            "continent": "varath",
            "faction_territory": "neutral",

        return {
            "zone": zone_meta,
            "rooms": [],
            "exits": [],
        }

class TestLoadZoneFromJsonBasic(ZoneSerializerTestBase):
    def test_minimal_zone_creates_zone_object(self):
        """load_zone_from_json() with only zone+rooms creates a ZoneObject."""
        data = self._minimal_zone_data("basic_zone")
        report = load_zone_from_json(data)
        self.assertEqual(report["zone_id"], "basic_zone")
        zone_obj = zone_registry.get_zone("basic_zone")
        self.assertIsNotNone(zone_obj)
        self.assertEqual(zone_obj.db.zone_id, "basic_zone")
        self.assertEqual(zone_obj.db.zone_type, "plains")

    def test_zone_with_rooms_creates_room_objects(self):
        """load_zone_from_json() with rooms list creates room DB objects."""
        data = self._minimal_zone_data("rooms_zone")
        data["rooms"] = [
            {"room_id": "r001", "name": "Open Plains", "desc": "Flat terrain.",
             "room_type": "path", "indoor": False, "terrain": "plains"},
            {"room_id": "r002", "name": "Rocky Outcrop", "desc": "Stones.",
             "room_type": "clearing", "indoor": False, "terrain": "rock"},
        ]
        report = load_zone_from_json(data)
        self.assertEqual(report["rooms_created"], 2)
        # Verify by tag lookup
        import evennia
        candidates = evennia.search_tag("r001", category="room_id")
        self.assertEqual(len(candidates), 1)
        self.assertEqual(candidates[0].db.zone_id, "rooms_zone")

    def test_zone_with_local_exit_creates_exit_object(self):
        """load_zone_from_json() with exits creates SoravelonExit between rooms."""
        from typeclasses.exits import SoravelonExit
        data = self._minimal_zone_data("exit_zone")
        data["rooms"] = [
            {"room_id": "r001", "name": "Room A", "desc": ".", "room_type": "path"},
            {"room_id": "r002", "name": "Room B", "desc": ".", "room_type": "path"},

        data["exits"] = [

        r1 = evennia.search_tag("r001", category="room_id")[0]
        exits = [ex for ex in r1.exits if ex.key == "north"]
        self.assertEqual(len(exits), 1)
        self.assertIsInstance(exits[0], SoravelonExit)

class TestLoadZoneFromJsonCrossZoneExit(ZoneSerializerTestBase):
    def test_cross_zone_exit_deferred_then_resolved(self):
        """JSON zone with cross-zone exit defers on first pass, resolves when target loads."""
        # Build target zone first (as .py would)
        ab_target = AreaBuilder("target_json_zone")
        ab_target.zone(
            name="Target Zone",
            tier=1,
            zone_type="plains",
            continent="varath",
            faction_territory="neutral",
        )
        import evennia
        from typeclasses.rooms import SoravelonRoom
        from evennia import create_object
        target_room = create_object(SoravelonRoom, key="Target Room", location=None)
        target_room.db.zone_id = "target_json_zone"
        target_room.tags.add("target_json_zone", category="zone_id")
        target_room.tags.add("target_r001", category="room_id")
        ab_target._rooms["target_r001"] = target_room
        ab_target.build()

        # Load JSON zone with cross-zone exit to target_json_zone:target_r001
        data = self._minimal_zone_data("source_json_zone")
        data["rooms"] = [{"room_id": "src_r001", "name": "Source", "desc": ".", "room_type": "path"}]

        self.assertEqual(len(report.get("unresolved_exits", [])), 0)
        src_rooms = evennia.search_tag("src_r001", category="room_id")
        src_room = next(r for r in src_rooms if r.db.zone_id == "source_json_zone")
        exits = [ex for ex in src_room.exits if ex.key == "east"]
        self.assertEqual(len(exits), 1)
        self.assertEqual(exits[0].destination, target_room)
