```python
"""
Mob spawner for Soravelon.

Reads spawn_definitions and named_mob_definitions from rooms and produces
live SoravelonMob instances with affixes, combat stats, and patrol scripts.

Public API:
    spawn_room_mobs(room)              — spawn all mobs defined on a room
    spawn_zone(zone_obj)               — spawn all mobs in a zone
    respawn_mob(room, spawn_def)       — respawn one mob for a definition (called by callLater)
    spawn_single_mob(mob_template, room) — create one mob from a template string
    spawn_named_mob(named_def, room)   — create a named mob and register it
"""

import random
import evennia
from evennia import create_object


def _room_has_no_mobs_flag(room):
    """Return True if room has the no_mobs tag (safe zone flag per D-04)."""
    return bool(room.tags.get("no_mobs", category="room_flag"))


def spawn_single_mob(mob_template, room):
    """
    Create one SoravelonMob in room from a mob template string.

    mob_template is the key string (e.g. "wolf", "bandit_scout").
    Sets mob.db.mob_type = mob_template and mob.db.zone_id from room.
    Calls initialize_for_spawn() for affixes + combat stats.

    Returns the created mob object.
    """
    from typeclasses.mobs import SoravelonMob

    mob = create_object(
        SoravelonMob,
        key=mob_template,
        location=room,
    )
    mob.db.mob_type = mob_template
    mob.db.zone_id = room.db.zone_id or ""
    mob.initialize_for_spawn(room)
    return mob


def _count_matching_mobs(room, mob_template):
    """Count SoravelonMob instances in room matching mob_template."""
    from typeclasses.mobs import SoravelonMob

    count = 0
    for obj in room.contents:
        if isinstance(obj, SoravelonMob) and (obj.db.mob_type or "") == mob_template:
            count += 1
    return count


def spawn_room_mobs(room):
    """
    Spawn all mobs defined on a room via room.db.spawn_definitions.

    Skips the room entirely if it has the no_mobs tag (D-04).
    For each spawn definition:
      - Count existing mobs matching the definition's mob template
      - If count < count_min, spawn up to count_min (per D-19)
      - Roll a target between count_min and count_max on first spawn (empty room)
    Also handles named_mob_definitions on the room (calls spawn_named_mob for each).
    """
    if _room_has_no_mobs_flag(room):
        return

    spawn_definitions = room.db.spawn_definitions or []
    for spawn_def in spawn_definitions:
        mob_template = spawn_def.get("mob")
        if not mob_template:
            continue

        count_min = spawn_def.get("count_min", 1)
        count_max = spawn_def.get("count_max", 1)
        existing = _count_matching_mobs(room, mob_template)

        if existing >= count_min:
            # Already at or above minimum — do not spawn (D-19)
            continue

        if existing == 0:
            # Fresh room — roll target count between min and max (D-18)
            target = random.randint(count_min, count_max)
        else:
            # Partially populated — fill up to count_min only
            target = count_min

        to_spawn = target - existing
        for _ in range(to_spawn):
            mob = spawn_single_mob(mob_template, room)
            # Apply spawn_def attributes to mob (D-09)
            mob.db.base_disposition = spawn_def.get("base_disposition", 0.0)
            mob.db.trust_sensitive = spawn_def.get("trust_sensitive", False)
            if spawn_def.get("flee_threshold"):
                mob.db.flee_threshold = spawn_def["flee_threshold"]

    # Handle named mob definitions on this room
    named_defs = room.db.named_mob_definitions or []
    for named_def in named_defs:
        mob_id = named_def.get("mob_id")
        if not mob_id:
            continue
        from world import named_mob_registry
        if named_mob_registry.get_named_mob(mob_id) is not None:
            # Already registered in this load cycle — skip creation
            continue
        # Check DB for existing named mob from previous server start (BLD-06 Blocker 2 fix)
        # Registry was cleared at start, but mob object may still exist in DB
        existing_in_room = [
            obj for obj in room.contents
            if hasattr(obj, 'db') and (obj.db.mob_type or "") == mob_id
        ]
        if existing_in_room:
            # Re-register existing mob without creating a duplicate
            named_mob_registry.register_named_mob(mob_id, room.db.zone_id or "", room, named_def)
            continue
        spawn_named_mob(named_def, room)


def spawn_named_mob(named_def, room):
    """
    Create a named mob instance and register it in named_mob_registry.

    Named mobs:
      - Are created with their mob_id as the key
      - Have tome_drop stored on db.tome_drop (pre-assigned at spawn per D-14)
      - Are registered in named_mob_registry (D-15)
      - Get PatrolScript if a matching patrol definition exists on the zone (D-16)

    Args:
        named_def: dict from room.db.named_mob_definitions
        room: Room object where the mob spawns

    Returns:
        The created SoravelonMob object.
    """
    from typeclasses.mobs import SoravelonMob
    from world import named_mob_registry

    mob_id = named_def["mob_id"]
    mob = create_object(
        SoravelonMob,
        key=mob_id,
        location=room,
    )
    mob.db.mob_type = mob_id
    mob.db.zone_id = room.db.zone_id or ""
    mob.db.tome_drop = named_def.get("tome_drop")
    mob.initialize_for_spawn(room)

    # Register in named mob registry (D-15)
    zone_id = room.db.zone_id or ""
    named_mob_registry.register_named_mob(mob_id, zone_id, room, named_def)

    # Attach patrol script if a patrol definition exists for this mob_id (D-16)
    _maybe_attach_patrol(mob, zone_id, named_def)

    return mob


def _maybe_attach_patrol(mob, zone_id, spawn_def):
    """
    Attach PatrolScript if a patrol definition in the zone matches this mob's mob_type/mob_id.

    Patrol definitions are stored on zone_obj.db.patrol_definitions (list of dicts).
    Each patrol_def has "mob_key" matching mob.db.mob_type or mob.key.
    """
    from world import zone_registry
    zone_obj = zone_registry.get_zone(zone_id)
    if not zone_obj:
        return

    patrol_defs = zone_obj.db.patrol_definitions or []
    mob_key = mob.db.mob_type or mob.key

    matching_patrol = None
    for pd in patrol_defs:
        if pd.get("mob_key") == mob_key:
            matching_patrol = pd
            break

    if not matching_patrol:
        return

    # Resolve route_room_ids to room dbrefs
    route_room_ids = matching_patrol.get("route_room_ids", [])
    route_dbrefs = []
    for room_id_str in route_room_ids:
        candidates = evennia.search_tag(room_id_str, category="room_id")
        for candidate in candidates:
            if (candidate.db.zone_id or "") == zone_id:
                route_dbrefs.append(candidate.id)
                break

    if not route_dbrefs:
        return

    from world.scripts.patrol_script import PatrolScript
    script = mob.scripts.add(PatrolScript)
    script.db.route_ids = route_dbrefs
    script.db.route_index = 0
    script.db.patrol_def = matching_patrol
    mob.db.patrol = matching_patrol


def spawn_zone(zone_obj):
    """
    Spawn all mobs in a zone by iterating every room tagged with zone_id.

    Called from _load_all_zones() after each zone's build() completes.
    Idempotent — spawn_room_mobs skips rooms already at count_min.

    Args:
        zone_obj: ZoneObject with db.zone_id set
    """
    zone_id = zone_obj.db.zone_id
    if not zone_id:
        return

    rooms = evennia.search_tag(zone_id, category="zone_id")
    for room in rooms:
        # Only spawn in rooms, not zone objects or exits
        from typeclasses.rooms import SoravelonRoom
        if isinstance(room, SoravelonRoom):
            spawn_room_mobs(room)


def respawn_mob(room, spawn_def):
    """
    Create one replacement mob for a spawn definition after a respawn timer fires.

    Called by Twisted callLater (one-shot, non-persistent per D-05 design note).
    Guards against count_max to prevent over-population (D-20).

    Args:
        room: Room object where mob should respawn
        spawn_def: The spawn definition dict from room.db.spawn_definitions
    """
    if not room or not room.pk:
        return  # Room deleted — abort

    if _room_has_no_mobs_flag(room):
        return

    mob_template = spawn_def.get("mob")
    if not mob_template:
        return

    count_max = spawn_def.get("count_max", 1)
    existing = _count_matching_mobs(room, mob_template)

    if existing >= count_max:
        return  # Already at cap — skip (D-20)

    mob = spawn_single_mob(mob_template, room)
    mob.db.base_disposition = spawn_def.get("base_disposition", 0.0)
    mob.db.trust_sensitive = spawn_def.get("trust_sensitive", False)
    if spawn_def.get("flee_threshold"):
        mob.db.flee_threshold = spawn_def["flee_threshold"]
```
