```python
def at_death(self, killer=None):
    """
    Clean up ndb on death, fire on_mob_death triggers, schedule respawn,
    and create tome drop for named mobs.
    """
    self.ndb.revealed_affixes = set()
    if hasattr(self.ndb, 'combat_scales'):
        self.ndb.combat_scales = {}
    if self.db.triggers:
        from world.trigger_engine import fire_triggers
        context = {"mob": self, "room": self.location}
        if killer and hasattr(killer, 'account') and killer.account:
            fire_triggers(self, "on_mob_death", killer, context=context)

    # Create tome drop before mob is deleted (D-14, D-26)
    tome_template_id = self.db.tome_drop
    if tome_template_id and self.location:
        try:
            from world.item_spawner import create_item_from_template
            create_item_from_template(tome_template_id, location=self.location)
        except Exception as _e:
            import traceback
            traceback.print_exc()

    # Schedule respawn timer (D-05, D-07)
    # Find which spawn_definition on the room matches this mob's mob_type
    room = self.location
    mob_type = self.db.mob_type
    if room and mob_type:
        spawn_definitions = room.db.spawn_definitions or []
        matching_def = None
        for spawn_def in spawn_definitions:
            if spawn_def.get("mob") == mob_type:
                matching_def = spawn_def
                break

        if matching_def:
            import random
            respawn_minutes = matching_def.get("respawn_minutes", 15)
            respawn_variance = matching_def.get("respawn_variance", 5)
            delay = (respawn_minutes * 60) + random.randint(
                -respawn_variance * 60, respawn_variance * 60
            )
            delay = max(delay, 5)  # minimum 5-second respawn
            from twisted.internet import reactor
            from world.mob_spawner import respawn_mob
            reactor.callLater(delay, respawn_mob, room, matching_def)
```

def _handle_spawn_mob(action_dict, context, _depth):
    """Spawn a mob from a template string in the current room."""
    room = context.get("room")
    if not room:
        character = context.get("character")
        room = character.location if character else None

    mob_template = action_dict.get("mob_template")
    if not mob_template:

    from world.mob_spawner import spawn_single_mob
    mob = spawn_single_mob(mob_template, room)
    return True, f"Spawned {mob.key} in {room.key}"
