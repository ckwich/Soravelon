```python
"""
Item template registry for Soravelon.

Item templates are the content-authoring surface for items (D-27).
Builders define items as template dicts; item_spawner instantiates them.

Templates are plain Python dicts. The registry is a module-level dict
rebuilt on every server start — no DB persistence required.

Template dict keys:
    key (str, required)          — Display name / Evennia object key
    item_type (str, required)    — "equipment", "container", "credential", or custom string
    weight (float)               — Item weight in kg (default 0.1)
    rarity (str)                 — "common", "uncommon", "rare", "legendary" (default "common")
    desc (str)                   — Room/inventory description
    lore_desc (str|None)         — Extended lore text shown on examine
    value_scales (int)           — Trade value in Scales currency (default 0)
    stackable (bool)             — Whether identical items stack (default False)

    equipment fields (item_type == "equipment"):
        equipment_slot (str)     — One of VALID_SLOTS: head/body/hands/feet/right_hand/left_hand/accessory
        stat_bonuses (dict)      — {stat_name: int} modifiers applied while equipped

    container fields (item_type == "container"):
        weight_reduction (int)   — % reduction applied to contents weight (0-100)
        weight_capacity (float)  — Maximum total item weight storable

    credential fields (item_type == "credential"):
        unlocks_exit_tag (str|None) — Exit lock_tag this key unlocks

Usage::

    from world.item_templates import get_template, register_template
    tmpl = get_template("iron_sword")
    register_template("custom_blade", {...})
"""


# Module-level registry. Area files and content modules populate this.
ITEM_TEMPLATES: dict = {}


def get_template(template_id: str) -> dict | None:
    """
    Return the template dict for template_id, or None if not found.

    Args:
        template_id: The string key identifying the template.

    Returns:
        dict or None
    """
    return ITEM_TEMPLATES.get(template_id)


def register_template(template_id: str, template_dict: dict) -> None:
    """
    Register or overwrite an item template.

    Args:
        template_id: Unique string key.
        template_dict: Template definition dict (see module docstring for keys).
    """
    ITEM_TEMPLATES[template_id] = dict(template_dict)
```

```python
"""
Item spawner for Soravelon.

Creates Evennia item objects from item template definitions.
This is the runtime bridge between content-authored item templates
and live game objects.

Public API:
    create_item_from_template(template_id, location=None) -> item_obj or None
"""

from evennia import create_object


def create_item_from_template(template_id: str, location=None):
    """
    Create a SoravelonItem (or subclass) from a registered item template.

    Dispatches to the correct typeclass based on template["item_type"]:
        "equipment"  -> SoravelonEquipment
        "container"  -> SoravelonContainer
        "credential" -> SoravelonKeyringItem
        other        -> SoravelonItem

    Sets all template fields on the created object's db attributes.
    location may be None (limbo) or any Evennia object (room, character, etc.).

    Args:
        template_id (str): Key into ITEM_TEMPLATES registry.
        location: Optional Evennia object for item placement.

    Returns:
        The created item object, or None if template_id not found.
    """
    from world.item_templates import get_template

    template = get_template(template_id)
    if template is None:
        return None

    item_type = template.get("item_type", "item")
    typeclass = _resolve_typeclass(item_type)

    item_key = template.get("key", template_id)
    item = create_object(typeclass, key=item_key, location=location)

    # Apply common item attributes
    item.db.weight = template.get("weight", 0.1)
    item.db.rarity = template.get("rarity", "common")
    item.db.desc = template.get("desc", "")
    item.db.lore_desc = template.get("lore_desc")
    item.db.value_scales = template.get("value_scales", 0)
    item.db.stackable = template.get("stackable", False)
    item.db.item_type = item_type

    # Tag for lookups
    item.tags.add(template_id, category="item_template_id")

    # Apply type-specific attributes
    if item_type == "equipment":
        item.db.equipment_slot = template.get("equipment_slot")
        item.db.stat_bonuses = dict(template.get("stat_bonuses") or {})

    elif item_type == "container":
        item.db.weight_reduction = template.get("weight_reduction", 0)
        item.db.weight_capacity = template.get("weight_capacity", 10.0)

    elif item_type == "credential":
        item.db.unlocks_exit_tag = template.get("unlocks_exit_tag")
        item.db.keyring = True

    return item


def _resolve_typeclass(item_type: str):
    """
    Return the correct item typeclass for item_type string.

    equipment  -> SoravelonEquipment
    container  -> SoravelonContainer
    credential -> SoravelonKeyringItem
    other      -> SoravelonItem
    """
    from typeclasses.objects import (
        SoravelonItem,
        SoravelonEquipment,
        SoravelonContainer,
        SoravelonKeyringItem,
    )

    dispatch = {
        "equipment": SoravelonEquipment,
        "container": SoravelonContainer,
        "credential": SoravelonKeyringItem,
    }
    return dispatch.get(item_type, SoravelonItem)
```
