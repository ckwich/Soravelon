    """Count enforcement: count_min and count_max respected."""

    def setUp(self):
        super().setUp()
        from typeclasses.rooms import SoravelonRoom
        self.room = create_object(SoravelonRoom, key="test_room", location=None)
        self.room.db.zone_id = "test_zone"
        self.room.db.named_mob_definitions = []

        super().tearDown()
        if self.room and self.room.pk:

                obj.delete()
            self.room.delete()

    def test_spawn_creates_count_min_mobs(self):
        """spawn_room_mobs creates at least count_min mobs when room is empty."""
        from world.mob_spawner import spawn_room_mobs
        from typeclasses.mobs import SoravelonMob
        self.room.db.spawn_definitions = [
            {"mob": "wolf", "count_min": 2, "count_max": 2,

             "base_disposition": 0.0, "trust_sensitive": False}
        ]

        mob_count = sum(
            1 for obj in self.room.contents

        )
        self.assertEqual(mob_count, 2)

        """spawn_room_mobs skips creation if existing mobs >= count_min (reload case)."""
        from world.mob_spawner import spawn_room_mobs, spawn_single_mob
        from typeclasses.mobs import SoravelonMob
        # Pre-create a wolf (simulates reload case)
        existing = spawn_single_mob("wolf", self.room)
        self.room.db.spawn_definitions = [
            {"mob": "wolf", "count_min": 1, "count_max": 3,
             "respawn_minutes": 15, "respawn_variance": 0,
             "base_disposition": 0.0, "trust_sensitive": False}
        ]
        spawn_room_mobs(self.room)
        mob_count = sum(
            1 for obj in self.room.contents
            if isinstance(obj, SoravelonMob) and obj.db.mob_type == "wolf"
        )
        # Still 1 — should not add more since existing >= count_min
        self.assertEqual(mob_count, 1)

    def test_respawn_mob_respects_count_max(self):
        """respawn_mob does not create mob if existing count >= count_max."""
        from world.mob_spawner import respawn_mob, spawn_single_mob
        from typeclasses.mobs import SoravelonMob
        spawn_def = {"mob": "wolf", "count_min": 1, "count_max": 1,
                     "respawn_minutes": 15, "respawn_variance": 0,
                     "base_disposition": 0.0, "trust_sensitive": False}
        self.room.db.spawn_definitions = [spawn_def]

        spawn_single_mob("wolf", self.room)
        respawn_mob(self.room, spawn_def)
        mob_count = sum(
            1 for obj in self.room.contents
            if isinstance(obj, SoravelonMob) and obj.db.mob_type == "wolf"
        )
        # Still 1 — count_max guard prevented extra spawn
        self.assertEqual(mob_count, 1)
