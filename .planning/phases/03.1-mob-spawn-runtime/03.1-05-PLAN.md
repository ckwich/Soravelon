```python
import random


# ---------------------------------------------------------------------------
# Loot tables — data-driven drop definitions (D-29)
# ---------------------------------------------------------------------------
# Keyed by mob_type string (primary) or zone_id (fallback).
# Each entry:
#   base_drops (list[str])       — template_ids always included in loot
#   bonus_pool (list[str])       — template_ids drawn during extra_rolls
#   bonus_drop_chance (float)    — probability per bonus_pool slot (0.0-1.0)
#   material_tier_override (int|None) — force tier instead of skill-derived (D-31)
#
# base_drops are guaranteed; bonus_pool is rolled once per extra_roll from rarity.
# Phase 3.1 ships with sample entries — content phase (Phase 7) populates fully.

LOOT_TABLES: dict = {
    # -----------------------------------------------------------------------
    # Starter entries — basic mob types
    # -----------------------------------------------------------------------
    "wolf": {
        "base_drops": [],
        "bonus_pool": ["healing_draught"],
        "bonus_drop_chance": 0.4,
        "material_tier_override": None,
    },
    "bandit": {
        "base_drops": [],
        "bonus_pool": ["healing_draught", "crude_iron_sword"],
        "bonus_drop_chance": 0.35,
        "material_tier_override": None,
    },
    "bandit_scout": {
        "base_drops": [],
        "bonus_pool": ["healing_draught"],
        "bonus_drop_chance": 0.3,
        "material_tier_override": None,
    },
    # -----------------------------------------------------------------------
    # Zone-level fallback — used when mob_type has no dedicated entry.
    # Zone IDs will be populated during Phase 7 content work.
    # -----------------------------------------------------------------------
    # "vael_crossing": {
    #     "base_drops": [],
    #     "bonus_pool": ["healing_draught"],
    #     "bonus_drop_chance": 0.2,
    #     "material_tier_override": None,
    # },
}


def roll_loot(mob, killer) -> list:
    """
    Resolve the loot list for a killed mob.

    Lookup priority (D-29):
      1. LOOT_TABLES[mob.db.mob_type]
      2. LOOT_TABLES[mob.db.zone_id]   (zone-level fallback)
      3. Return [] (safe no-op — D-30 says items land in room, so nothing = no drop)

    extra_rolls are determined by mob rarity via LOOT_TIER_MODIFIERS.
    Material tier is from killer's primary skill or material_tier_override (D-31).

    Args:
        mob:    The dying SoravelonMob.
        killer: The character that killed the mob (may be None for environment deaths).

    Returns:
        list of template_id strings. May be empty.
    """
    # Resolve table entry
    mob_type = mob.db.mob_type or ""
    zone_id = mob.db.zone_id or ""

    table_entry = LOOT_TABLES.get(mob_type) or LOOT_TABLES.get(zone_id)
    if not table_entry:
        return []

    drops = []

    # Guaranteed base drops
    drops.extend(table_entry.get("base_drops") or [])

    # Bonus drops — rolled per extra_roll from rarity modifiers
    modifiers = get_loot_modifiers(mob)
    extra_rolls = modifiers.get("extra_rolls", 0)
    bonus_pool = table_entry.get("bonus_pool") or []
    bonus_chance = table_entry.get("bonus_drop_chance", 0.3)

    if bonus_pool and extra_rolls > 0:
        for _ in range(extra_rolls):
            if random.random() < bonus_chance:
                drops.append(random.choice(bonus_pool))

    return drops
```
