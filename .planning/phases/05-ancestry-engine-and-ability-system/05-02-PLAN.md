```python
# Build DOMAIN_ABILITIES: {domain: {tier: [ability_ids]}}
DOMAIN_ABILITIES = {}
for ability_id, ability in ABILITIES.items():
    if ability["subclass_id"] is not None:
        continue  # signatures are separate
    domain = ability["domain"]
    tier = ability["tier"]
    DOMAIN_ABILITIES.setdefault(domain, {}).setdefault(tier, []).append(ability_id)

# Build SUBCLASS_SIGNATURES: {subclass_id: [ability_ids]}
SUBCLASS_SIGNATURES = {}
for ability_id, ability in ABILITIES.items():
    if ability["subclass_id"]:
        sc = ability["subclass_id"]
        SUBCLASS_SIGNATURES.setdefault(sc, []).append(ability_id)
```

**world/models.py** — Add CharacterAbility model per D-11. Place after CharacterGuild class (end of file).

```python
class CharacterAbility(models.Model):
    """
    Tracks which abilities a character has unlocked.

    Source of truth for ability access. Created when player claims an ability
    at guild hall (D-19). Queried by ability engine for access checks.
    """

    character = models.ForeignKey(
        "objects.ObjectDB",
        on_delete=models.CASCADE,
        related_name="character_abilities",
    )
    ability_id = models.CharField(max_length=128, db_index=True)
    unlocked_at = models.DateTimeField(auto_now_add=True)
    times_used = models.IntegerField(default=0)

    class Meta:
        unique_together = ("character", "ability_id")
        indexes = [
            models.Index(fields=["character_id", "ability_id"]),
        ]

    def __str__(self):
        return f"{self.character.db_key}:{self.ability_id}"
```
